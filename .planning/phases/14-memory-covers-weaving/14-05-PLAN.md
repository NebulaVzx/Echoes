```tsx
'use client'

import { useState } from 'react'
import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog'
import { Button } from '@/components/ui/button'
import { FileText, BookOpen, ListChecks, ScrollText, Loader2 } from 'lucide-react'

export type WeaveMode = 'article' | 'story' | 'summary' | 'todo'

interface WeaveModalProps {
  open: boolean
  onOpenChange: (open: boolean) => void
  sourceIds: string[]
  onWeave: (mode: WeaveMode, title?: string) => Promise<void>
  isLoading?: boolean
}

const MODES: { id: WeaveMode; label: string; description: string; icon: typeof FileText }[] = [
  { id: 'article', label: '文章', description: '将记忆编织成一篇连贯的文章', icon: FileText },
  { id: 'story', label: '故事', description: '用叙事方式串联记忆', icon: BookOpen },
  { id: 'summary', label: '摘要', description: '提炼核心要点，按主题归类', icon: ScrollText },
  { id: 'todo', label: '任务清单', description: '转化为可执行的行动项', icon: ListChecks },

export function WeaveModal({ open, onOpenChange, sourceIds, onWeave, isLoading }: WeaveModalProps) {
  const [selectedMode, setSelectedMode] = useState<WeaveMode>('article')
  const [title, setTitle] = useState('')

  const handleWeave = async () => {

      <DialogContent className="sm:max-w-md">
        <DialogHeader>
          <DialogTitle>编织文章 — 已选择 {sourceIds.length} 条记忆</DialogTitle>

          {/* Mode selection */}

              return (
                <button
                  key={mode.id}
                  onClick={() => setSelectedMode(mode.id)}
                  className={`flex flex-col items-center gap-2 p-4 rounded-xl border-2 transition-all ${
                    selectedMode === mode.id
                      ? 'border-primary bg-primary/5'
                      : 'border-gray-200 dark:border-gray-700 hover:border-gray-300'
                  }`}
                >
                  <Icon className="w-6 h-6 text-primary" />
                  <span className="text-sm font-medium">{mode.label}</span>
```

```tsx
'use client'

import { useEffect, useState } from 'react'
import { useRouter } from 'next/navigation'
import { WeaveModal, WeaveMode } from '@/components/weave/weave-modal'
import { api } from '@/lib/api'
import { Loader2 } from 'lucide-react'

export default function WeavePage() {
  const router = useRouter()
  const [sourceIds, setSourceIds] = useState<string[]>([])
  const [isLoading, setIsLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    const stored = sessionStorage.getItem('weave_source_ids')
    if (stored) {
      try {
        const ids = JSON.parse(stored)
        if (Array.isArray(ids) && ids.length >= 2) {
          setSourceIds(ids)
        } else {
          setError('请至少选择 2 条记忆进行编织')
        }
      } catch {
        setError('无效的选择数据')
      }
    } else {
      setError('未选择记忆，请从时间轴选择后编织')
    }
  }, [])

  const handleWeave = async (mode: WeaveMode, title?: string) => {
    setIsLoading(true)
    setError(null)
    try {
      const response = await api.createWeave({
        source_ids: sourceIds,
        mode,
        title,
      })
      if (response.success && response.data?.memory) {
        sessionStorage.removeItem('weave_source_ids')
        router.push(`/weave/${response.data.memory.id}`)
      } else {
        setError(response.error?.message || '编织失败')
      }
    } catch (err) {
      setError(err instanceof Error ? err.message : '编织失败')
    } finally {
      setIsLoading(false)
    }
  }

  if (error) {
    return (
      <div className="mx-auto max-w-content-timeline px-4 py-12 text-center">
        <p className="text-red-500">{error}</p>
        <button
          onClick={() => router.push('/')}
          className="mt-4 text-primary hover:underline"
        >
          返回时间轴
        </button>
      </div>
    )
  }

  if (sourceIds.length === 0) {
    return (
      <div className="mx-auto max-w-content-timeline px-4 py-12 flex justify-center">
        <Loader2 className="w-8 h-8 animate-spin text-primary" />
      </div>
    )
  }

  return (
    <WeaveModal
      open={true}
      onOpenChange={() => router.push('/')}
      sourceIds={sourceIds}
      onWeave={handleWeave}
      isLoading={isLoading}
    />
  )
}
```

```tsx
'use client'

import { useEffect, useState } from 'react'
import { useParams, useRouter } from 'next/navigation'
import { api, Memory } from '@/lib/api'
import { WeaveEditor } from '@/components/weave/weave-editor'
import { Loader2, ArrowLeft } from 'lucide-react'

export default function WeaveDetailPage() {
  const params = useParams()
  const router = useRouter()
  const id = params.id as string

  const [memory, setMemory] = useState<Memory | null>(null)
  const [isLoading, setIsLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    const fetchMemory = async () => {
      try {
        const response = await api.getMemory(id)
        if (response.success && response.data) {
          setMemory(response.data)
        } else {
          setError('记忆不存在')
        }
      } catch (err) {
        setError(err instanceof Error ? err.message : '加载失败')
      } finally {
        setIsLoading(false)
      }
    }
    fetchMemory()
  }, [id])

  const handleSave = async (content: string) => {
    try {
      const response = await api.updateMemory(id, { text_content: content })
      if (response.success) {
        setMemory(prev => prev ? { ...prev, text_content: content } : null)
        return true
      }
      return false
    } catch {
      return false
    }
  }

  if (isLoading) {
    return (
      <div className="mx-auto max-w-content-timeline px-4 py-12 flex justify-center">
        <Loader2 className="w-8 h-8 animate-spin text-primary" />
      </div>
    )
  }

  if (error || !memory) {
    return (
      <div className="mx-auto max-w-content-timeline px-4 py-12 text-center">
        <p className="text-red-500">{error || '记忆不存在'}</p>
      </div>
    )
  }

  return (
    <div className="mx-auto max-w-content-timeline px-4 py-6">
      <button
        onClick={() => router.push('/')}
        className="flex items-center gap-1 text-sm text-gray-500 hover:text-gray-700 mb-4"
      >
        <ArrowLeft className="w-4 h-4" />
        返回时间轴
      </button>

      <WeaveEditor
        memory={memory}
        onSave={handleSave}
      />
    </div>
  )
}
```

```tsx
'use client'

import { useState, useCallback } from 'react'
import { Memory } from '@/lib/api'
import { Button } from '@/components/ui/button'
import { Textarea } from '@/components/ui/textarea'
import { Download, Save, Check, FileEdit } from 'lucide-react'

interface WeaveEditorProps {
  memory: Memory
  onSave: (content: string) => Promise<boolean>
}

export function WeaveEditor({ memory, onSave }: WeaveEditorProps) {
  const [content, setContent] = useState(memory.text_content || '')
  const [isEditing, setIsEditing] = useState(false)
  const [isSaving, setIsSaving] = useState(false)
  const [saveSuccess, setSaveSuccess] = useState(false)

  const handleSave = async () => {
    setIsSaving(true)
    setSaveSuccess(false)
    const success = await onSave(content)
    setIsSaving(false)
    if (success) {
      setSaveSuccess(true)
      setIsEditing(false)
      setTimeout(() => setSaveSuccess(false), 2000)
    }
  }

  const handleExport = useCallback(() => {
    const blob = new Blob([content], { type: 'text/markdown;charset=utf-8' })
    const url = window.URL.createObjectURL(blob)
    const link = document.createElement('a')
    link.href = url
    const title = memory.text_content?.split('\n')[0]?.replace(/^#+\s*/, '') || 'weave'
    link.download = `${title.slice(0, 50)}.md`
    document.body.appendChild(link)
    link.click()
    document.body.removeChild(link)
    window.URL.revokeObjectURL(url)
  }, [content, memory.text_content])

  // Parse metadata for source info
  let sourceIds: string[] = []
  let weaveMode = ''
  try {
    const meta = JSON.parse(memory.metadata || '{}')
    sourceIds = meta.weave_source_ids || []
    weaveMode = meta.weave_mode || ''
  } catch {
    // ignore
  }

  const modeLabels: Record<string, string> = {
    article: '文章',
    story: '故事',
    summary: '摘要',
    todo: '任务清单',
  }

  return (
    <div className="space-y-4">
      {/* Header */}
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-xl font-semibold">
            {memory.text_content?.split('\n')[0]?.replace(/^#+\s*/, '').slice(0, 100) || '编织文章'}
          </h1>
          {weaveMode && (
            <span className="text-xs text-gray-500 bg-gray-100 dark:bg-gray-800 px-2 py-0.5 rounded-full">
              {modeLabels[weaveMode] || weaveMode}
            </span>
          )}
        </div>
        <div className="flex items-center gap-2">
          {isEditing ? (
            <>
              <Button
                variant="outline"
                size="sm"
                onClick={() => {
                  setContent(memory.text_content || '')
                  setIsEditing(false)
                }}
              >
                取消
              </Button>
              <Button
                size="sm"
                onClick={handleSave}
                disabled={isSaving}
              >
                {isSaving ? '保存中...' : saveSuccess ? (
                  <><Check className="w-4 h-4 mr-1" />已保存</>
                ) : (
                  <><Save className="w-4 h-4 mr-1" />保存</>
                )}
              </Button>
            </>
          ) : (
            <>
              <Button variant="outline" size="sm" onClick={() => setIsEditing(true)}>
                <FileEdit className="w-4 h-4 mr-1" />
                编辑
              </Button>
              <Button variant="outline" size="sm" onClick={handleExport}>
                <Download className="w-4 h-4 mr-1" />
                导出 Markdown
              </Button>
            </>
          )}
        </div>
      </div>

      {/* Content */}
      {isEditing ? (
        <Textarea
          value={content}
          onChange={(e) => setContent(e.target.value)}
          className="min-h-[500px] font-mono text-sm"
          placeholder="编辑编织内容..."
        />
      ) : (
        <div className="prose dark:prose-invert max-w-none">
          <pre className="whitespace-pre-wrap font-sans text-sm leading-relaxed bg-transparent p-0">
            {content}
          </pre>
        </div>
      )}

      {/* Source info */}
      {sourceIds.length > 0 && (
        <div className="pt-4 border-t border-gray-200 dark:border-gray-700">
          <p className="text-xs text-gray-500">
            来源记忆: {sourceIds.length} 条
          </p>
        </div>
      )}
    </div>
  )
}
```
