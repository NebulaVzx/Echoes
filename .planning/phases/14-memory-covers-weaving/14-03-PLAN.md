创建 `web/lib/tag-color.ts`，提取星座图中使用的标签 hash 颜色逻辑：

```typescript
/**
 * Generate a consistent HSL color from a tag string.
 * Uses MD5-like hashing (simple djb2) to produce hue 0-360.
 * Matches the color algorithm used in constellation graph.
 */
export function getTagColor(tag: string): string {
  let hash = 5381
  for (let i = 0; i < tag.length; i++) {
    hash = ((hash << 5) + hash) + tag.charCodeAt(i)
    hash = hash & 0xffffffff
  }
  const hue = Math.abs(hash % 360)
  return `hsl(${hue}, 65%, 55%)`
}

/**
 * Generate a gradient background string from a tag color.
 */
export function getTagGradient(tag: string): string {
  const color = getTagColor(tag)
  return `linear-gradient(135deg, ${color}, hsl(${parseInt(color.match(/\d+/)?.[0] || '0') + 30}, 65%, 45%))`
}

/**
 * Get the first letter of a tag for fallback display.
 */
export function getTagInitial(tag: string): string {
  return tag.charAt(0).toUpperCase()
}
```

注意：此函数必须与星座图使用的颜色算法一致。如果星座图使用不同的 hash 算法，需要调整以匹配。当前假设使用简单的 djb2 hash → hue mod 360。
