```typescript
import { useRef, useCallback } from 'react'

export function useLongPress(callback: () => void, threshold = 500) {
  const timerRef = useRef<NodeJS.Timeout | null>(null)

  const start = useCallback((e: React.TouchEvent | React.MouseEvent) => {
    if ('touches' in e) {
      // Touch event — prevent default to avoid context menu
      // Don't preventDefault here; do it in onContextMenu
    }
    timerRef.current = setTimeout(() => {
      callback()
      timerRef.current = null
    }, threshold)
  }, [callback, threshold])

  const cancel = useCallback(() => {
    if (timerRef.current) {
      clearTimeout(timerRef.current)
      timerRef.current = null
    }
  }, [])

  return {
    onTouchStart: start,
    onTouchEnd: cancel,
    onTouchMove: cancel,
    onContextMenu: (e: React.MouseEvent) => e.preventDefault(),
  }
}
```

2. 实现选择切换逻辑：
```tsx
const handleSelectToggle = useCallback((e: React.MouseEvent, memoryId: string, index: number) => {
  if (e.ctrlKey || e.metaKey) {
    // Ctrl/Cmd + click: toggle individual
    setSelectedIds(prev => {
      const next = new Set(prev)
      if (next.has(memoryId)) {
        next.delete(memoryId)
      } else {
        next.add(memoryId)
      }
      return next
    })
    setLastSelectedId(memoryId)
    if (!isSelectionMode) setIsSelectionMode(true)
  } else if (e.shiftKey && lastSelectedId !== null) {
    // Shift + click: range select
    const lastIndex = memories.findIndex(m => m.id === lastSelectedId)
    if (lastIndex === -1) {
      // Anchor lost, treat as single select
      setSelectedIds(new Set([memoryId]))
      setLastSelectedId(memoryId)
      if (!isSelectionMode) setIsSelectionMode(true)
      return
    }
    const [start, end] = [lastIndex, index].sort((a, b) => a - b)
    const rangeIds = memories.slice(start, end + 1).map(m => m.id)
    setSelectedIds(prev => {
      const next = new Set(prev)
      rangeIds.forEach(id => next.add(id))
      return next
    })
    setLastSelectedId(memoryId)
    if (!isSelectionMode) setIsSelectionMode(true)
  } else {
    // Normal click in selection mode: toggle
    if (isSelectionMode) {
      setSelectedIds(prev => {
        const next = new Set(prev)
        if (next.has(memoryId)) {
          next.delete(memoryId)
        } else {
          next.add(memoryId)
        }
        return next
      })
      setLastSelectedId(memoryId)
    }
  }
}, [memories, lastSelectedId, isSelectionMode])
```
