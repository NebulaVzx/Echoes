    1. Create `services/processor-service/app/observability.py`:
       ```python
       from contextlib import asynccontextmanager
       from fastapi import FastAPI
       from prometheus_client import make_asgi_app, Counter, Histogram
       from opentelemetry import trace
       from opentelemetry.sdk.trace import TracerProvider
       from opentelemetry.sdk.resources import Resource
       from opentelemetry.sdk.trace.export import BatchSpanProcessor
       from opentelemetry.exporter.otlp.proto.http.trace_exporter import OTLPSpanExporter
       from opentelemetry.instrumentation.fastapi import FastAPIInstrumentor
       import os
       import logging

       # Prometheus metrics (per D-06)
       http_requests_total = Counter(
           'http_requests_total',
           'Total HTTP requests',
           ['service', 'method', 'path', 'status_code']
       )
       http_request_duration_seconds = Histogram(
           'http_request_duration_seconds',
           'HTTP request duration in seconds',
           ['service', 'method', 'path']
       )

       def setup_observability(app: FastAPI, service_name: str):
           """Setup Prometheus metrics and OpenTelemetry tracing."""
           # 1. Prometheus /metrics endpoint
           metrics_app = make_asgi_app()
           app.mount("/metrics", metrics_app)

           # 2. OpenTelemetry TracerProvider
           resource = Resource.create({
               "service.name": service_name,
               "service.version": os.getenv("SERVICE_VERSION", "0.2.0"),
           })
           provider = TracerProvider(resource=resource)

           # OTLP HTTP exporter -> Jaeger
           otlp_endpoint = os.getenv("OTEL_EXPORTER_OTLP_ENDPOINT", "http://jaeger:4318/v1/traces")
           exporter = OTLPSpanExporter(endpoint=otlp_endpoint)
           provider.add_span_processor(BatchSpanProcessor(exporter))

           trace.set_tracer_provider(provider)

           # 3. Auto instrument FastAPI
           FastAPIInstrumentor.instrument_app(app)

           # 4. Configure structured logging
           log_level = os.getenv("LOG_LEVEL", "INFO")
           logging.basicConfig(
               level=getattr(logging, log_level),
               format='{"timestamp": "%(asctime)s", "level": "%(levelname)s", "msg": "%(message)s", "service": "' + service_name + '", "trace_id": "%(otelTraceID)s", "span_id": "%(otelSpanID)s"}'
           )

           return provider
       ```
       Note: The logging format uses %(otelTraceID)s which requires a custom filter or the opentelemetry-logging integration. Simpler approach: use standard logging and add trace_id manually where needed.
       SIMPLIFIED logging approach:
       ```python
       logging.basicConfig(
           level=getattr(logging, os.getenv("LOG_LEVEL", "INFO")),
           format='%(asctime)s [%(levelname)s] %(name)s: %(message)s'
       )
       ```
       Then in consumer code, manually add trace_id to log messages when span is active.

         ```python
             carrier = {"traceparent": traceparent}
             parent_context = extract(carrier)

             parent_context = None

         with tracer.start_as_current_span(
             "process_redis_task",
             context=parent_context,
             attributes={
                 "stream": self.stream,
                 "message_id": msg_id,
                 "memory_id": memory_id,
             }
         ) as span:
             # Existing retry loop goes here
             for attempt in range(retry_count, self.max_retries):
                 try:
                     await self.process_message(msg_id, fields)
                     await self.redis.xack(self.stream, self.group, msg_id)
                     span.set_attribute("success", True)
                     logger.info(f"Successfully processed {self.stream} for memory {memory_id}")
                     return
                 except Exception as e:
                     span.set_attribute("error", True)
                     span.set_attribute("error.message", str(e))
                     logger.error(f"Error processing {self.stream} for memory {memory_id} (attempt {attempt + 1}/{self.max_retries}): {e}")
                     # ... rest of error handling
         ```
