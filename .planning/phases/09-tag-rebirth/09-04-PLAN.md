    Create web/components/tag/tag-cloud.tsx:

    ```tsx
    'use client'

    import { TagStats } from '@/lib/api'
    import Link from 'next/link'

    interface TagCloudProps {
      tags: TagStats[]
      tagColors: Record<string, string>
    }

    export default function TagCloud({ tags, tagColors }: TagCloudProps) {
      if (tags.length === 0) {
        return (
          <div className="text-center py-12">
            <p className="text-sm text-gray-400 dark:text-gray-500">还没有标签，保存第一条记忆后会自动生成</p>
          </div>
        )
      }

      const maxCount = Math.max(...tags.map((t) => t.memory_count), 1)

      return (
        <div className="flex flex-wrap items-center justify-center gap-3 py-8">
          {tags.map((tag) => {
            const ratio = tag.memory_count / maxCount
            const fontSize = Math.max(12, Math.min(28, 12 + ratio * 16))
            const color = tagColors[tag.name] || '#f3f4f6'
            return (
              <Link
                key={tag.name}
                href={`/?tags=${encodeURIComponent(tag.name)}`}
                className="px-3 py-1.5 rounded-full transition-all hover:scale-105 hover:shadow-sm"
                style={{
                  fontSize: `${fontSize}px`,
                  backgroundColor: color,
                  color: '#374151',
                }}
              >
                #{tag.name}
              </Link>
            )
          })}
        </div>
      )
    }
    ```

## Task 3: Create TagMergeDialog component

<task>
  <id>09-04-03</id>
  <title>Create TagMergeDialog for merging similar tags</title>
  <read_first>
    - web/lib/api.ts -- mergeTags method
  </read_first>
  <action>
    Create web/components/tag/tag-merge-dialog.tsx:

    ```tsx
    'use client'

    import { useState } from 'react'
    import { api, TagStats } from '@/lib/api'

    interface TagMergeDialogProps {
      tags: TagStats[]
      isOpen: boolean
      onClose: () => void
      onMerged: () => void
    }

    export default function TagMergeDialog({ tags, isOpen, onClose, onMerged }: TagMergeDialogProps) {
      const [sourceTag, setSourceTag] = useState('')
      const [targetTag, setTargetTag] = useState('')
      const [isLoading, setIsLoading] = useState(false)

      if (!isOpen) return null

      const handleMerge = async () => {
        if (!sourceTag || !targetTag || sourceTag === targetTag) return
        setIsLoading(true)
        try {
          const res = await api.mergeTags(sourceTag, targetTag)
          if (res.success) {
            onMerged()
            onClose()
          }
        } catch {
          // Ignore error
        } finally {
          setIsLoading(false)
        }
      }

      // Auto-detect similar tags (case-insensitive match)
      const similarPairs = []
      const normalized = tags.map((t) => ({ original: t.name, lower: t.name.toLowerCase().trim() }))
      for (let i = 0; i < normalized.length; i++) {
        for (let j = i + 1; j < normalized.length; j++) {
          if (normalized[i].lower === normalized[j].lower && normalized[i].original !== normalized[j].original) {
            similarPairs.push([normalized[i].original, normalized[j].original])
          }
        }
      }

      return (
        <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50">
          <div className="bg-white dark:bg-gray-800 rounded-xl p-6 w-full max-w-md mx-4">
            <h3 className="text-lg font-medium text-gray-900 dark:text-gray-100 mb-4">合并标签</h3>

            {similarPairs.length > 0 && (
              <div className="mb-4 p-3 bg-amber-50 dark:bg-amber-900/20 rounded-lg">
                <p className="text-xs text-amber-700 dark:text-amber-300 mb-2">检测到相似标签:</p>
                {similarPairs.map(([a, b]) => (
                  <button
                    key={`${a}-${b}`}
                    className="block text-xs text-amber-600 dark:text-amber-400 hover:underline mb-1"
                    onClick={() => { setSourceTag(a); setTargetTag(b) }}
                  >
                    将 "{a}" 合并到 "{b}"
                  </button>
                ))}
              </div>
            )}

            <div className="space-y-3 mb-4">
              <div>
                <label className="block text-sm text-gray-600 dark:text-gray-400 mb-1">源标签 (将被替换)</label>
                <select
                  value={sourceTag}
                  onChange={(e) => setSourceTag(e.target.value)}
                  className="w-full px-3 py-2 bg-gray-50 dark:bg-gray-700 border border-gray-200 dark:border-gray-600 rounded-md text-sm"
                >
                  <option value="">选择标签...</option>
                  {tags.map((t) => (
                    <option key={t.name} value={t.name}>{t.name}</option>
                  ))}
                </select>
              </div>
              <div>
                <label className="block text-sm text-gray-600 dark:text-gray-400 mb-1">目标标签</label>
                <select
                  value={targetTag}
                  onChange={(e) => setTargetTag(e.target.value)}
                  className="w-full px-3 py-2 bg-gray-50 dark:bg-gray-700 border border-gray-200 dark:border-gray-600 rounded-md text-sm"
                >
                  <option value="">选择标签...</option>
                  {tags.map((t) => (
                    <option key={t.name} value={t.name}>{t.name}</option>
                  ))}
                </select>
              </div>
            </div>

            <div className="flex justify-end gap-2">
              <button
                onClick={onClose}
                className="px-4 py-2 text-sm text-gray-600 dark:text-gray-400 hover:bg-gray-100 dark:hover:bg-gray-700 rounded-md transition-colors"
              >取消</button>
              <button
                onClick={handleMerge}
                disabled={!sourceTag || !targetTag || sourceTag === targetTag || isLoading}
                className="px-4 py-2 text-sm bg-gray-900 dark:bg-gray-100 text-white dark:text-gray-900 rounded-md hover:bg-gray-800 dark:hover:bg-gray-200 transition-colors disabled:opacity-50"
              >
                {isLoading ? '合并中...' : '合并'}
              </button>
            </div>
          </div>
        </div>
      )
    }
    ```
  </action>
  <acceptance_criteria>
    - web/components/tag/tag-merge-dialog.tsx exists
    - Auto-detects case-different similar tags
    - Shows detected pairs as quick-select buttons
    - Has source and target tag dropdowns
    - Calls api.mergeTags on confirm
    - Disabled when source === target or either is empty
  </acceptance_criteria>
</task>
