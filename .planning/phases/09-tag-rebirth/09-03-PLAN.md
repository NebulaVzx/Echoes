    ```typescript
    async listMemories(params?: { page?: number; limit?: number; tag?: string; tags?: string[] }): Promise<ApiResponse<ListMemoriesResponse>> {
      const searchParams = new URLSearchParams()
      if (params?.page) searchParams.set('page', String(params.page))
      if (params?.limit) searchParams.set('limit', String(params.limit))
      if (params?.tag) searchParams.set('tag', params.tag)
      if (params?.tags) {
        params.tags.forEach(tag => searchParams.append('tags', tag))
      }
      const query = searchParams.toString()
      return this.request<ListMemoriesResponse>('GET', `/api/v1/memories${query ? '?' + query : ''}`)
    }
    ```

    3. Add tag API methods:
    ```typescript
    async getAllTags(): Promise<ApiResponse<{ tags: TagStats[] }>> {
      return this.request<{ tags: TagStats[] }>('GET', '/api/v1/tags')
    }

    async getRelatedTags(tagName: string, limit?: number): Promise<ApiResponse<{ related_tags: RelatedTag[] }>> {
      const searchParams = new URLSearchParams()
      if (limit) searchParams.set('limit', String(limit))
      const query = searchParams.toString()
      return this.request<{ related_tags: RelatedTag[] }>('GET', `/api/v1/tags/${encodeURIComponent(tagName)}/related${query ? '?' + query : ''}`)
    }

    async mergeTags(sourceTag: string, targetTag: string): Promise<ApiResponse<{ affected_memories: number }>> {
      return this.request<{ affected_memories: number }>('POST', '/api/v1/tags/merge', {
        source_tag: sourceTag,
        target_tag: targetTag,
      })
    }
    ```

<task>

    Create web/components/tag/tag-filter-bar.tsx:

    ```tsx
    'use client'

    import { useState, useEffect, useCallback } from 'react'
    import { api, TagStats } from '@/lib/api'
    import { X } from 'lucide-react'

    interface TagFilterBarProps {
      selectedTags: string[]
      onChange: (tags: string[]) => void
    }

    const TAG_COLORS = [
      '#E8F4FD', '#E6F5E6', '#FFF4E6', '#F3E8FF', '#FFE4E6',
      '#E0F7FA', '#F5F5DC', '#E8F5E9', '#FFF8E1', '#FCE4EC',
      '#E3F2FD', '#E0F2F1', '#F9FBE7', '#EDE7F6', '#ECEFF1', '#FBE9E7',
    ]

    export default function TagFilterBar({ selectedTags, onChange }: TagFilterBarProps) {
      const [allTags, setAllTags] = useState<TagStats[]>([])
      const [isLoading, setIsLoading] = useState(true)

      useEffect(() => {
        api.getAllTags().then((res) => {
          if (res.success && res.data) {
            setAllTags(res.data.tags.slice(0, 12)) // Show top 12 tags
          }
        }).catch(() => {}).finally(() => setIsLoading(false))
      }, [])

      const toggleTag = useCallback((tagName: string) => {
        if (selectedTags.includes(tagName)) {
          onChange(selectedTags.filter((t) => t !== tagName))
        } else {
          onChange([...selectedTags, tagName])
        }
      }, [selectedTags, onChange])

      const clearAll = useCallback(() => {
        onChange([])
      }, [onChange])

      if (isLoading) return null
      if (allTags.length === 0) return null

      return (
        <div className="flex flex-wrap items-center gap-2 mb-5">
          {allTags.map((tag, index) => {
            const isSelected = selectedTags.includes(tag.name)
            const bgColor = TAG_COLORS[index % TAG_COLORS.length]
            return (
              <button
                key={tag.name}
                onClick={() => toggleTag(tag.name)}
                className={`inline-flex items-center gap-1 px-3 py-1 text-xs rounded-full transition-colors ${
                  isSelected
                    ? 'bg-gray-900 dark:bg-gray-100 text-white dark:text-gray-900'
                    : 'text-gray-600 dark:text-gray-400 hover:bg-gray-100 dark:hover:bg-gray-700'
                }`}
                style={isSelected ? {} : { backgroundColor: bgColor }}
              >
                #{tag.name}
                {isSelected && <X className="w-3 h-3" />}
              </button>
            )
          })}
          {selectedTags.length > 0 && (
            <button
              onClick={clearAll}
              className="text-xs text-gray-400 dark:text-gray-500 hover:text-gray-600 dark:hover:text-gray-300 transition-colors"
            >
              清除全部
            </button>
          )}
        </div>
      )
    }
    ```
  </action>
  <acceptance_criteria>
    - web/components/tag/tag-filter-bar.tsx exists
    - Fetches tags via api.getAllTags()
    - Displays top 12 tags as clickable pills
    - Selected state: dark bg + white text
    - Unselected state: colored bg from preset palette
    - Shows "清除全部" when tags are selected
    - Calls onChange with updated tag array on click
  </acceptance_criteria>
</task>
