    package repository

    import (
      "context"
      "time"

      "github.com/NebulaVzx/Echoes/services/memory-service/internal/domain"
      "github.com/google/uuid"
      "gorm.io/gorm"
    )

    // TagRepository defines the interface for tag data access.
    type TagRepository interface {
      GetAllTags(ctx context.Context, userID uuid.UUID) ([]domain.TagStats, error)
      GetRelatedTags(ctx context.Context, userID uuid.UUID, tagName string, limit int) ([]domain.RelatedTag, error)
      MergeTags(ctx context.Context, userID uuid.UUID, sourceTag, targetTag string) (int64, error)
    }

    // GormTagRepository implements TagRepository using GORM.
    type GormTagRepository struct {
      db *gorm.DB
    }

    // NewGormTagRepository creates a new GORM-based tag repository.
    func NewGormTagRepository(db *gorm.DB) TagRepository {
      return &GormTagRepository{db: db}
    }

    // GetAllTags retrieves all unique tags for a user with usage statistics.
    func (r *GormTagRepository) GetAllTags(ctx context.Context, userID uuid.UUID) ([]domain.TagStats, error) {
      var stats []domain.TagStats
      query := `
        SELECT
          tag as name,
          COUNT(*) as memory_count,
          MAX(created_at) as last_used_at
        FROM memories, UNNEST(tags) as tag
        WHERE user_id = ?
        GROUP BY tag
        ORDER BY memory_count DESC, name ASC
      `
      err := r.db.WithContext(ctx).Raw(query, userID).Scan(&stats).Error
      return stats, err
    }

    // GetRelatedTags finds tags that co-occur most frequently with the given tag.
    func (r *GormTagRepository) GetRelatedTags(ctx context.Context, userID uuid.UUID, tagName string, limit int) ([]domain.RelatedTag, error) {
      var related []domain.RelatedTag
      query := `
        SELECT
          t2.tag as name,
          COUNT(*) as cooccurrence
        FROM memories m1
        JOIN memories m2 ON m1.id = m2.id
        CROSS JOIN UNNEST(m1.tags) as t1(tag)
        CROSS JOIN UNNEST(m2.tags) as t2(tag)
        WHERE m1.user_id = ?
          AND t1.tag = ?
          AND t2.tag != ?
        GROUP BY t2.tag
        ORDER BY cooccurrence DESC, t2.tag ASC
        LIMIT ?
      `
      err := r.db.WithContext(ctx).Raw(query, userID, tagName, tagName, limit).Scan(&related).Error
      return related, err
    }

    // MergeTags replaces sourceTag with targetTag in all of the user's memories.
    func (r *GormTagRepository) MergeTags(ctx context.Context, userID uuid.UUID, sourceTag, targetTag string) (int64, error) {
      result := r.db.WithContext(ctx).Exec(`
        UPDATE memories
        SET tags = ARRAY(
          SELECT CASE WHEN x = ? THEN ? ELSE x END
          FROM UNNEST(tags) AS x
        ),
        updated_at = ?
        WHERE user_id = ?
          AND ? = ANY(tags)
      `, sourceTag, targetTag, time.Now(), userID, sourceTag)
      if result.Error != nil {
        return 0, result.Error
      }
      return result.RowsAffected, nil
    }
    ```
  </action>

    - GetAllTags uses UNNEST(tags) aggregation

</task>

    func (r *GormMemoryRepository) ListByUser(ctx context.Context, userID uuid.UUID, page, limit int, tags []string) ([]domain.Memory, int64, error) {
      var memories []domain.Memory
      var total int64

      query := r.db.WithContext(ctx).Model(&domain.Memory{}).Where("user_id = ?", userID)
      if len(tags) > 0 {
        query = query.Where("tags @> ?", pq.Array(tags))
      }

      if err := query.Count(&total).Error; err != nil {
        return nil, 0, err
      }

      offset := (page - 1) * limit
      result := query.Order("created_at DESC").Offset(offset).Limit(limit).Find(&memories)
      if result.Error != nil {
        return nil, 0, result.Error
      }

      return memories, total, nil
    }

    package service

    import (
      "context"
      "strings"

      "github.com/NebulaVzx/Echoes/services/memory-service/internal/domain"
      "github.com/NebulaVzx/Echoes/services/memory-service/internal/repository"
      "github.com/google/uuid"
    )

    // TagService handles tag-related business logic.
    type TagService struct {
      tagRepo repository.TagRepository
    }

    // NewTagService creates a new TagService.
    func NewTagService(tagRepo repository.TagRepository) *TagService {
      return &TagService{tagRepo: tagRepo}
    }

    // GetAllTags retrieves all tags with statistics for a user.
    func (s *TagService) GetAllTags(ctx context.Context, userID uuid.UUID) ([]domain.TagStats, error) {
      return s.tagRepo.GetAllTags(ctx, userID)
    }

    // GetRelatedTags retrieves tags that co-occur with the given tag.
    func (s *TagService) GetRelatedTags(ctx context.Context, userID uuid.UUID, tagName string, limit int) ([]domain.RelatedTag, error) {
      if limit <= 0 || limit > 20 {
        limit = 3
      }
      return s.tagRepo.GetRelatedTags(ctx, userID, tagName, limit)
    }

    // MergeTags merges sourceTag into targetTag for all memories of the user.
    func (s *TagService) MergeTags(ctx context.Context, userID uuid.UUID, req domain.MergeTagsRequest) (int64, error) {
      source := strings.ToLower(strings.TrimSpace(req.SourceTag))
      target := strings.ToLower(strings.TrimSpace(req.TargetTag))
      if source == target {
        return 0, ErrInvalidRequest
      }
      return s.tagRepo.MergeTags(ctx, userID, source, target)
    }
    ```

  </action>

    - MergeTags normalizes to lowercase and trims whitespace
    - MergeTags returns error if source == target
