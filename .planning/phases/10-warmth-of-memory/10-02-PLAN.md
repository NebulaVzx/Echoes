    func (r *GormMemoryRepository) ListSealedMemories(ctx context.Context, userID uuid.UUID, page, limit int) ([]domain.Memory, int64, error) {
      var memories []domain.Memory
      var total int64

      query := r.db.WithContext(ctx).Model(&domain.Memory{}).
        Where("user_id = ?", userID).
        Where("sealed_until > ?", time.Now())

      if err := query.Count(&total).Error; err != nil {
        return nil, 0, err
      }

      offset := (page - 1) * limit
      err := query.Order("sealed_until ASC").Offset(offset).Limit(limit).Find(&memories).Error
      return memories, total, err
    }

    Add to `MemoryService`:
    ```go
    func (s *MemoryService) SealMemory(ctx context.Context, userID, memoryID uuid.UUID, sealedUntil time.Time) error {
      if sealedUntil.Before(time.Now()) {
        return ErrInvalidRequest // Cannot seal to the past
      }
      return s.memoryRepo.SealMemory(ctx, userID, memoryID, sealedUntil)
    }

    func (s *MemoryService) UnsealMemory(ctx context.Context, userID, memoryID uuid.UUID) error {
      return s.memoryRepo.UnsealMemory(ctx, userID, memoryID)
    }

    func (s *MemoryService) ListSealedMemories(ctx context.Context, userID uuid.UUID, page, limit int) (*domain.ListMemoriesResponse, error) {
      memories, total, err := s.memoryRepo.ListSealedMemories(ctx, userID, page, limit)
      if err != nil {
        return nil, err
      }
      // Build response using SafeResponse for each memory
      result := &domain.ListMemoriesResponse{
        Total: total,
        Page:  page,
        Limit: limit,
      }
      for _, m := range memories {
        result.Memories = append(result.Memories, m.SafeResponse())
      }
      result.HasMore = total > int64(page*limit)
      return result, nil
    }

    func (s *MemoryService) GetRecentlyUnsealed(ctx context.Context, userID uuid.UUID) ([]domain.Memory, error) {
      // Check for memories that became unsealed in the last 24 hours
      since := time.Now().Add(-24 * time.Hour)
      return s.memoryRepo.GetRecentlyUnsealed(ctx, userID, since)
    }
    ```
  </action>
  <acceptance_criteria>
    - MemoryService.SealMemory rejects past dates
    - MemoryService.UnsealMemory passes through to repository
    - ListSealedMemories returns paginated response with HasMore
    - GetRecentlyUnsealed checks last 24 hours
  </acceptance_criteria>
</task>

    func (h *MemoryHandler) SealMemory(c *gin.Context) {
      userID := getUserID(c)
      memoryID, err := uuid.Parse(c.Param("id"))
      if err != nil {
        respondWithError(c, 400, "INVALID_ID", "Invalid memory ID")
        return
      }

      var req domain.SealMemoryRequest
      if err := c.ShouldBindJSON(&req); err != nil {
        respondWithValidationError(c, err)
        return
      }

      if err := h.memoryService.SealMemory(c.Request.Context(), userID, memoryID, req.SealedUntil); err != nil {
        respondWithError(c, 400, "SEAL_FAILED", err.Error())
        return
      }

      c.JSON(200, gin.H{"success": true})
    }

    func (h *MemoryHandler) UnsealMemory(c *gin.Context) {
      userID := getUserID(c)
      memoryID, err := uuid.Parse(c.Param("id"))
      if err != nil {
        respondWithError(c, 400, "INVALID_ID", "Invalid memory ID")
        return
      }

      if err := h.memoryService.UnsealMemory(c.Request.Context(), userID, memoryID); err != nil {
        respondWithError(c, 400, "UNSEAL_FAILED", err.Error())
        return
      }

      c.JSON(200, gin.H{"success": true})
    }

    func (h *MemoryHandler) ListSealedMemories(c *gin.Context) {
      userID := getUserID(c)
      page := 1
      limit := 20
      // Parse page/limit from query

      resp, err := h.memoryService.ListSealedMemories(c.Request.Context(), userID, page, limit)
      if err != nil {
        respondWithError(c, 500, "QUERY_FAILED", err.Error())
        return
      }

      c.JSON(200, gin.H{"success": true, "data": resp})
    }

    func (h *MemoryHandler) GetRecentlyUnsealed(c *gin.Context) {
      userID := getUserID(c)
      memories, err := h.memoryService.GetRecentlyUnsealed(c.Request.Context(), userID)
      if err != nil {
        respondWithError(c, 500, "QUERY_FAILED", err.Error())
        return
      }

      var results []map[string]interface{}
      for _, m := range memories {
        results = append(results, m.SafeResponse())
      }

      c.JSON(200, gin.H{"success": true, "data": gin.H{"memories": results}})
    }

    async listSealedMemories(params?: { page?: number; limit?: number }): Promise<ApiResponse<ListMemoriesResponse>> {
