```typescript
'use client'

import { useState, useEffect, useCallback, useRef } from 'react'
import { useRouter } from 'next/navigation'
import { useAuth } from '@/app/providers/auth-provider'
import { useTheme } from '@/app/providers/theme-provider'
import {
  CommandDialog,
  CommandEmpty,
  CommandGroup,
  CommandInput,
  CommandItem,
  CommandList,
  CommandSeparator,
} from '@/components/ui/command'
import {
  Clock,
  Sparkles,
  Compass,
  Tag,
  Smile,
  Bell,
  PenLine,
  Package2,
  User,
  Settings,
  Sun,
  Moon,
  LogOut,
  Search,
  Plus,
  Shuffle,
} from 'lucide-react'

// Navigation items for Command Palette — mirrors Sidebar but adds extra commands
const NAV_ITEMS: { label: string; icon: typeof Clock; path: string; group: string }[] = [
  // 拾忆
  { label: '时间轴', icon: Clock, path: '/', group: 'pages' },
  { label: '记忆星图', icon: Sparkles, path: '/constellation', group: 'pages' },
  { label: '探索模式', icon: Compass, path: '/explore', group: 'pages' },
  // 发现
  { label: '标签云', icon: Tag, path: '/tags', group: 'pages' },
  { label: '情绪日历', icon: Smile, path: '/mood', group: 'pages' },
  { label: '每日回响', icon: Bell, path: '/daily-echo', group: 'pages' },
  // 创作
  { label: '记忆编织', icon: PenLine, path: '/weave', group: 'pages' },
  { label: '时间胶囊', icon: Package2, path: '/capsules', group: 'pages' },
  // 我的
  { label: '个人画像', icon: User, path: '/profile', group: 'pages' },
  { label: '设置', icon: Settings, path: '/settings', group: 'pages' },
]

// Command actions ( > prefix )
const COMMANDS: { label: string; icon: typeof Clock; action: string; group: string }[] = [
  { label: '切换主题', icon: Sun, action: 'theme', group: 'commands' },
  { label: '随机探索', icon: Shuffle, action: 'explore', group: 'commands' },
]

export function CommandPalette() {
  const [open, setOpen] = useState(false)
  const [query, setQuery] = useState('')
  const router = useRouter()
  const { logout } = useAuth()
  const { toggleTheme, resolvedTheme } = useTheme()
  const inputRef = useRef<HTMLInputElement>(null)

  // Cmd+K / Ctrl+K keyboard shortcut
  useEffect(() => {
    const down = (e: KeyboardEvent) => {
      if (e.key === 'k' && (e.metaKey || e.ctrlKey)) {
        e.preventDefault()
        setOpen((prev) => !prev)
      }
    }
    document.addEventListener('keydown', down)
    return () => document.removeEventListener('keydown', down)
  }, [])

  // Reset query when opening/closing
  useEffect(() => {
    if (open) {
      setQuery('')
      // Focus input after animation frame
      requestAnimationFrame(() => {
        inputRef.current?.focus()
      })
    }
  }, [open])

  const runCommand = useCallback((fn: () => unknown) => {
    setOpen(false)
    fn()
  }, [])

  // Detect prefix modes
  const isSlashMode = query.startsWith('/')
  const isCommandMode = query.startsWith('>')
  const trimmedQuery = isSlashMode || isCommandMode ? query.slice(1).trim() : query

  // Filter items based on query
  const filteredNavItems = NAV_ITEMS.filter(
    item => item.label.toLowerCase().includes(trimmedQuery.toLowerCase())
  )

  // Execute command on select
  const handleCommandSelect = (action: string) => {
    switch (action) {
      case 'theme':
        toggleTheme()
        setOpen(false)
        break
      case 'explore':
        router.push('/explore')
        setOpen(false)
        break
    }
  }

  return (
    <CommandDialog open={open} onOpenChange={setOpen}>
      <CommandInput
        ref={inputRef}
        placeholder="搜索记忆或输入命令..."
        value={query}
        onValueChange={setQuery}
      />

      <CommandList>
        <CommandEmpty>没有找到匹配项</CommandEmpty>

        {/* Slash mode: filter hints */}
        {isSlashMode && (
          <CommandGroup heading="筛选方式">
            <CommandItem onSelect={() => setQuery('/标签 ')}>
              <Tag className="mr-2 h-4 w-4" />
              按标签筛选
            </CommandItem>
            <CommandItem onSelect={() => setQuery('/时间 ')}>
              <Clock className="mr-2 h-4 w-4" />
              按时间筛选
            </CommandItem>
          </CommandGroup>
        )}

        {/* Command mode: available commands */}
        {isCommandMode && (
          <CommandGroup heading="可用命令">
            {COMMANDS.map(cmd => (
              <CommandItem
                key={cmd.action}
                value={cmd.label}
                onSelect={() => handleCommandSelect(cmd.action)}
              >
                <cmd.icon className="mr-2 h-4 w-4" />
                {cmd.label}
              </CommandItem>
            ))}
          </CommandGroup>
        )}

        {/* Normal mode: navigation + quick actions */}
        {!isSlashMode && !isCommandMode && (
          <>
            {/* Page navigation */}
            <CommandGroup heading="页面导航">
              {filteredNavItems.map(item => (
                <CommandItem
                  key={item.path}
                  value={item.label}
                  onSelect={() => runCommand(() => router.push(item.path))}
                >
                  <item.icon className="mr-2 h-4 w-4" />
                  {item.label}
                </CommandItem>
              ))}
            </CommandGroup>

            <CommandSeparator />

            {/* Quick actions */}
            <CommandGroup heading="快捷操作">
              <CommandItem
                value="主题切换"
                onSelect={() => runCommand(() => toggleTheme())}
              >
                {resolvedTheme === 'dark' ? (
                  <Sun className="mr-2 h-4 w-4" />
                ) : (
                  <Moon className="mr-2 h-4 w-4" />
                )}
                切换主题
              </CommandItem>
              <CommandItem
                value="搜索记忆"
                onSelect={() => runCommand(() => router.push('/search'))}
              >
                <Search className="mr-2 h-4 w-4" />
                搜索记忆
              </CommandItem>
              <CommandItem
                value="打开设置"
                onSelect={() => runCommand(() => router.push('/settings'))}
              >
                <Settings className="mr-2 h-4 w-4" />
                打开设置
              </CommandItem>
            </CommandGroup>

            <CommandSeparator />

            {/* Hint footer */}
            <CommandGroup heading="提示">
              <div className="px-2 py-1.5 text-xs text-muted-foreground">
                输入 <kbd className="px-1 py-0.5 rounded bg-muted text-[10px] font-mono">/</kbd> 按标签筛选，输入{' '}
                <kbd className="px-1 py-0.5 rounded bg-muted text-[10px] font-mono">&gt;</kbd> 执行命令
              </div>
            </CommandGroup>
          </>
        )}
      </CommandList>
    </CommandDialog>
  )
}
```

Key behaviors:
- Cmd+K (Mac) / Ctrl+K (Win/Linux) toggles the palette
- Placeholder text: "搜索记忆或输入命令..." (per UI-SPEC)
- No results text: "没有找到匹配项" (per UI-SPEC)
- `/` prefix: shows category filter hints (按标签筛选, 按时间筛选)
- `>` prefix: shows available commands (切换主题, 随机探索)
- Normal mode: shows page navigation items with fuzzy search
- Hint footer shows keyboard shortcuts for `/` and `>` modes
- Uses `kbd` HTML element for keyboard shortcut display (styled inline with muted background)
- CommandDialog renders as a portal overlay — no z-index issues

Note: The `ref` prop on CommandInput may not be directly supported by shadcn's cmdk wrapper. If TypeScript complains, use a wrapper div approach or the `cmdk` `commandRef` prop. Check the actual shadcn command component exports before finalizing.
