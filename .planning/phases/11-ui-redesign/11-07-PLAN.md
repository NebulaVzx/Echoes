**Part B: Create web/components/layout/RightPanel.tsx**

```typescript
'use client'

import { useMemo, useCallback, useRef, useEffect } from 'react'
import { usePathname } from 'next/navigation'
import { useLayout } from '@/app/providers/layout-provider'
import { AnimatePresence, motion } from 'framer-motion'
import { RightPanelWidget } from '@/components/layout/RightPanelWidget'
import { ScrollArea } from '@/components/ui/scroll-area'
import { ChevronLeft, ChevronRight, GripVertical } from 'lucide-react'

export function RightPanel() {
  const pathname = usePathname()
  const { rightPanelVisible, rightPanelWidth, setRightPanelWidth, toggleRightPanel } = useLayout()
  const resizeRef = useRef<HTMLDivElement>(null)
  const isResizing = useRef(false)

  // Determine panel title and empty state based on pathname
  const panelContext = useMemo(() => {
    if (pathname === '/') {
      return { title: '时间轴助手', empty: '那年今日卡片、热门标签将显示在这里' }
    }
    if (pathname.startsWith('/memory/')) {
      return { title: '相关推荐', empty: '相似记忆和标签将在浏览详情时显示' }
    }
    if (pathname === '/search') {
      return { title: '搜索工具', empty: '搜索历史和筛选器将在这里显示' }
    }
    if (pathname === '/tags') {
      return { title: '标签分析', empty: '标签统计和关联将在这里显示' }
    }
    if (pathname === '/constellation') {
      return { title: '星图控制', empty: '筛选器和节点详情将在这里显示' }
    }
    if (pathname === '/explore') {
      return { title: '探索路径', empty: '探索轨迹和收藏的记忆将在这里显示' }
    }
    if (pathname === '/weave') {
      return { title: '编织素材', empty: '已选记忆列表和导出选项将在这里显示' }
    }
    if (pathname === '/mood') {
      return { title: '情绪洞察', empty: '月度情绪摘要和排行将在这里显示' }
    }
    if (pathname === '/capsules') {
      return { title: '胶囊状态', empty: '即将解锁的胶囊将在这里显示' }
    }
    if (pathname === '/settings') {
      return { title: '设置导航', empty: '快速跳转到各设置分类' }
    }
    return { title: '上下文面板', empty: '' }
  }, [pathname])

  // Resize drag handler
  const handleResizeStart = useCallback((e: React.MouseEvent) => {
    e.preventDefault()
    isResizing.current = true
    const startX = e.clientX
    const startWidth = rightPanelWidth

    const onMouseMove = (e: MouseEvent) => {
      if (!isResizing.current) return
      const delta = startX - e.clientX
      const newWidth = Math.min(400, Math.max(200, startWidth + delta))
      setRightPanelWidth(newWidth)
    }

    const onMouseUp = () => {
      isResizing.current = false
      document.removeEventListener('mousemove', onMouseMove)
      document.removeEventListener('mouseup', onMouseUp)
    }

    document.addEventListener('mousemove', onMouseMove)
    document.addEventListener('mouseup', onMouseUp)
  }, [rightPanelWidth, setRightPanelWidth])

  return (
    <AnimatePresence mode="wait">
      {rightPanelVisible ? (
        <motion.aside
          key="panel-open"
          initial={{ opacity: 0, x: 20 }}
          animate={{ opacity: 1, x: 0 }}
          exit={{ opacity: 0, x: 20 }}
          transition={{ duration: 0.2, ease: 'easeOut' }}
          className="right-panel flex flex-col h-full"
          style={{ width: rightPanelWidth }}
          data-visible="true"
        >
          {/* Resize handle (left edge) */}
          <div
            ref={resizeRef}
            onMouseDown={handleResizeStart}
            className="absolute left-0 top-0 bottom-0 w-1 cursor-col-resize hover:bg-primary/20 transition-colors z-10"
            title="拖拽调整宽度"
          >
            <GripVertical className="absolute top-1/2 -translate-y-1/2 left-1/2 -translate-x-1/2 h-4 w-4 text-muted-foreground/30" />
          </div>

          {/* Panel header */}
          <div className="flex items-center justify-between px-4 py-3 border-b border-border">
            <h2 className="text-sm font-medium text-foreground">{panelContext.title}</h2>
            <button
              onClick={toggleRightPanel}
              className="p-1 rounded-md text-muted-foreground hover:bg-muted hover:text-foreground transition-colors"
              aria-label="关闭面板"
            >
              <ChevronRight className="h-4 w-4" />
            </button>
          </div>

          {/* Panel content */}
          <ScrollArea className="flex-1">
            <div className="p-4 flex flex-col gap-3">
              {/* Page-specific widget content goes here — for now, show empty state */}
              <RightPanelWidget title={panelContext.title}>
                <p className="text-sm text-muted-foreground">
                  {panelContext.empty || '继续探索，相关内容会出现在这里'}
                </p>
              </RightPanelWidget>
            </div>
          </ScrollArea>
        </motion.aside>
      ) : (
        <motion.div
          key="panel-closed"
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          exit={{ opacity: 0 }}
          className="flex items-start justify-center pt-3 h-full border-l border-border bg-background"
        >
          <button
            onClick={toggleRightPanel}
            className="p-1 rounded-md text-muted-foreground hover:bg-muted hover:text-foreground transition-colors"
            aria-label="打开面板"
          >
            <ChevronLeft className="h-4 w-4" />
          </button>
        </motion.div>
      )}
    </AnimatePresence>
  )
}
```

```typescript
'use client'

import Link from 'next/link'
import { usePathname } from 'next/navigation'
import { Clock, Sparkles, Package2, Tag, User } from 'lucide-react'
import { cn } from '@/lib/utils'

const DOCK_ITEMS: { href: string; label: string; icon: typeof Clock; exact?: boolean }[] = [
  { href: '/',              label: '拾', icon: Clock,    exact: true },
  { href: '/constellation', label: '星', icon: Sparkles },
  { href: '/capsules',      label: '胶', icon: Package2  },
  { href: '/tags',          label: '标', icon: Tag       },
  { href: '/profile',       label: '我', icon: User      },
]

export function MobileDock() {
  const pathname = usePathname()

  return (
    <nav className="mobile-dock md:hidden" aria-label="移动端导航">
      <div className="flex items-center justify-around h-full max-w-md mx-auto">
        {DOCK_ITEMS.map((item) => {
          const isActive = item.exact ? pathname === item.href : pathname.startsWith(item.href)

          return (
            <Link
              key={item.href}
              href={item.href}
              className={cn(
                'flex flex-col items-center justify-center gap-0.5 min-w-0 flex-1 h-full',
                'text-muted-foreground hover:text-foreground transition-colors',
                isActive && 'text-primary'
              )}
            >
              <item.icon className="h-5 w-5" />
              <span className="text-[10px] leading-tight">{item.label}</span>
              {/* Active indicator dot */}
              {isActive && (
                <span className="absolute bottom-0 left-1/2 -translate-x-1/2 w-1 h-1 rounded-full bg-primary" />
              )}
            </Link>
          )
        })}
      </div>
    </nav>
  )
}
```

The MobileDock is ONLY visible on mobile via the `md:hidden` Tailwind class (hidden at 768px+).
The CSS for `.mobile-dock` is already defined in globals.css (Plan 02) with:
