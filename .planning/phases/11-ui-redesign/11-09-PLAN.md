```typescript
'use client'

import { useEffect, useRef, useCallback } from 'react'

interface Particle {
  x: number
  y: number
  vx: number
  vy: number
  life: number
  maxLife: number
  size: number
  color: string
}

const COLORS = ['#3b82f6', '#8b5cf6', '#ec4899', '#f59e0b', '#10b981', '#ef4444', '#6366f1']
const PARTICLE_COUNT = 60
const DURATION_MS = 500

interface ConfettiLiteProps {
  /** Whether to trigger the animation. Resets automatically after animation completes. */
  trigger: boolean
  /** Called when animation finishes */
  onComplete?: () => void
}

export function ConfettiLite({ trigger, onComplete }: ConfettiLiteProps) {
  const canvasRef = useRef<HTMLCanvasElement>(null)
  const animFrameRef = useRef<number>(0)

  const animate = useCallback(() => {
    const canvas = canvasRef.current
    if (!canvas) return

    const ctx = canvas.getContext('2d')
    if (!ctx) return

    const particles: Particle[] = []
    const startTime = Date.now()

    // Initialize particles from center-bottom
    const cx = canvas.width / 2
    const cy = canvas.height

    for (let i = 0; i < PARTICLE_COUNT; i++) {
      particles.push({
        x: cx + (Math.random() - 0.5) * 40,
        y: cy,
        vx: (Math.random() - 0.5) * 4 + (Math.random() - 0.5) * 2,
        vy: -(Math.random() * 5 + 2),
        life: 1,
        maxLife: 0.7 + Math.random() * 0.3,
        size: 3 + Math.random() * 4,
        color: COLORS[Math.floor(Math.random() * COLORS.length)],
      })
    }

    function frame() {
      const elapsed = Date.now() - startTime
      const progress = elapsed / DURATION_MS

      if (progress >= 1) {
        ctx.clearRect(0, 0, canvas.width, canvas.height)
        onComplete?.()
        return
      }

      ctx.clearRect(0, 0, canvas.width, canvas.height)

      for (const p of particles) {
        // Fade out near end of life
        const alpha = Math.max(0, p.life)
        p.x += p.vx
        p.y += p.vy
        p.vy += 0.08 // gravity
        p.life = 1 - progress / p.maxLife

        if (p.life > 0) {
          ctx.save()
          ctx.globalAlpha = alpha * 0.8
          ctx.fillStyle = p.color
          ctx.beginPath()
          // Draw small rectangle particles
          ctx.rect(p.x, p.y, p.size, p.size * 0.6)
          ctx.fill()
          ctx.restore()
        }
      }

      animFrameRef.current = requestAnimationFrame(frame)
    }

    animFrameRef.current = requestAnimationFrame(frame)
  }, [onComplete])

  useEffect(() => {
    if (!trigger) return

    const canvas = canvasRef.current
    if (!canvas) return

    // Size canvas to viewport
    canvas.width = window.innerWidth
    canvas.height = window.innerHeight

    // Cancel any running animation
    if (animFrameRef.current) {
      cancelAnimationFrame(animFrameRef.current)
    }

    animate()

    return () => {
      if (animFrameRef.current) {
        cancelAnimationFrame(animFrameRef.current)
      }
    }
  }, [trigger, animate])

  if (!trigger) return null

  return (
    <canvas
      ref={canvasRef}
      className="fixed inset-0 z-[100] pointer-events-none"
      aria-hidden="true"
    />
  )
}
```

Key behaviors:
- Triggered by `trigger` prop — parent component sets it to `true` on milestone, canvas renders
- 60 particles float up from center-bottom, affected by gravity, fading out
- Duration: ~500ms (per CONTEXT.md: "~500ms, non-blocking")
- Pointer-events: none — doesn't block user interaction
- Z-index: 100 — renders above everything
- 7 particle colors from Colors array
- Calls `onComplete` when animation finishes for cleanup
- Self-cleaning: returns null when not triggered

```typescript
'use client'

import { Flame } from 'lucide-react'
import { cn } from '@/lib/utils'

interface StreakFlameProps {
  count: number
  className?: string
}

export function StreakFlame({ count, className }: StreakFlameProps) {
  if (count < 1) return null

  // Flame color intensifies with streak length
  const flameClass = count >= 30
    ? 'text-purple-500 animate-pulse'
    : count >= 7
    ? 'text-orange-500 animate-pulse'
    : 'text-amber-500'

  return (
    <span className={cn('inline-flex items-center gap-1', className)} title={`连续 ${count} 天记录`}>
      <Flame className={cn('h-4 w-4', flameClass)} />
      <span className="text-xs font-medium tabular-nums">{count}</span>
    </span>
  )
}
```

```typescript
'use client'

import { useThemeColor } from '@/app/providers/theme-color-provider'
import { cn } from '@/lib/utils'
import { Check } from 'lucide-react'

type AccentPreset = 'neutral' | 'blue' | 'green' | 'orange' | 'violet'

const ACCENT_PRESETS: { id: AccentPreset; label: string; color: string; ring: string }[] = [
  { id: 'neutral', label: '石墨灰', color: '#1A1A1A', ring: 'ring-neutral-500' },
  { id: 'blue',    label: '深海蓝', color: '#3B82F6', ring: 'ring-blue-500' },
  { id: 'green',   label: '森林绿', color: '#10B981', ring: 'ring-emerald-500' },
  { id: 'orange',  label: '珊瑚橙', color: '#F97316', ring: 'ring-orange-500' },
  { id: 'violet',  label: '紫罗兰', color: '#8B5CF6', ring: 'ring-violet-500' },
]

export function ThemeColorPicker() {
  const { accent, setAccent } = useThemeColor()

  return (
    <div className="flex flex-col gap-2">
      <p className="text-sm font-medium text-foreground">主题色</p>
      <div className="flex gap-2">
        {ACCENT_PRESETS.map((preset) => {
          const isActive = accent === preset.id
          return (
            <button
              key={preset.id}
              onClick={() => setAccent(preset.id)}
              className={cn(
                'relative w-9 h-9 rounded-full border-2 transition-all duration-150',
                'hover:scale-110 focus:outline-none focus:ring-2 focus:ring-offset-2',
                isActive ? 'border-primary ring-2 ring-primary ring-offset-2' : 'border-transparent'
              )}
              style={{ backgroundColor: preset.color }}
              title={preset.label}
              aria-label={`选择 ${preset.label} 主题色`}
              aria-pressed={isActive}
            >
              {isActive && (
                <Check
                  className="absolute inset-0 m-auto h-4 w-4 text-white drop-shadow-sm"
                  strokeWidth={3}
                />
              )}
            </button>
          )
        })}
      </div>
      <p className="text-xs text-muted-foreground mt-1">
        当前: {ACCENT_PRESETS.find(p => p.id === accent)?.label}
      </p>
    </div>
  )
}
```

Design notes:
- StreakFlame: Shows Flame icon from lucide-react + count. Color transitions: amber (<7 days), orange (7-29), purple (30+). `animate-pulse` for gentle glow effect.
- ThemeColorPicker: 5 circular color swatches. Active swatch gets checkmark icon + ring highlight. Click calls `setAccent()` which instantly applies the accent via DOM API (ThemeColorProvider). Hover scales 110%. Displays current accent name below.
- Both are 'use client' components (interactivity required).
