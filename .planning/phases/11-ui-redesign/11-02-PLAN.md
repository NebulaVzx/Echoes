```css
/* ===== Phase 11: Three-Column Layout Shell ===== */

/* Layout CSS custom properties — defaults for desktop */
:root {
  --sidebar-width: 200px;
  --panel-width: 0px;
  --header-height: 48px;

  /* Density (comfortable is default) */
  --card-padding: 16px;
  --element-spacing: 16px;
}

/* Right panel visible state */
:root[data-panel-visible="true"] {
  --panel-width: 280px;
}

/* Sidebar collapsed state */
:root[data-sidebar-collapsed="true"] {
  --sidebar-width: 48px;
}

/* Density: compact */
:root[data-density="compact"] {
  --card-padding: 12px;
  --element-spacing: 8px;
}

/* Density: loose */
:root[data-density="loose"] {
  --card-padding: 24px;
  --element-spacing: 24px;
}

/* ===== AppShell Grid Container ===== */

.app-shell {
  display: grid;
  grid-template-rows: var(--header-height) 1fr;
  grid-template-columns: var(--sidebar-width) 1fr var(--panel-width);
  min-height: 100dvh;
  transition: grid-template-columns 200ms ease-out;
}

.app-shell-header {
  grid-column: 1 / -1;
  grid-row: 1;
}

.app-shell-sidebar {
  grid-row: 2;
  overflow: hidden;
  min-width: 0;
}

.app-shell-main {
  grid-row: 2;
  min-width: 0;
  overflow-y: auto;
}

.app-shell-panel {
  grid-row: 2;
  overflow: hidden;
  min-width: 0;
}

/* When right panel is hidden (0px track), hide content to prevent overflow */
.app-shell-panel[data-visible="false"] {
  visibility: hidden;
  pointer-events: none;
}

/* ===== Responsive Breakpoints ===== */

/* Tablet: two columns, no right panel */
@media (max-width: 1279px) {
  :root {
    --panel-width: 0px;
  }

  .app-shell {
    grid-template-columns: var(--sidebar-width) 1fr 0px;
  }
}

/* Tablet: sidebar as overlay drawer when collapsed */
@media (max-width: 1023px) {
  :root {
    --sidebar-width: 0px;
  }

  .app-shell {
    grid-template-columns: 0px 1fr 0px;
  }
}

/* Mobile: single column, dock at bottom */
@media (max-width: 767px) {
  .app-shell {
    grid-template-columns: 1fr;
    padding-bottom: calc(56px + env(safe-area-inset-bottom, 0px));
  }

  .app-shell-sidebar,
  .app-shell-panel {
    display: none;
  }
}

/* ===== Mobile Dock ===== */

.mobile-dock {
  position: fixed;
  bottom: 0;
  left: 0;
  right: 0;
  height: calc(56px + env(safe-area-inset-bottom, 0px));
  padding-bottom: env(safe-area-inset-bottom, 0px);
  background-color: hsl(var(--background));
  border-top: 1px solid hsl(var(--border));
  z-index: 50;
}

/* Safe-area fallback for older browsers */
@supports not (padding-bottom: env(safe-area-inset-bottom, 0px)) {
  .mobile-dock {
    padding-bottom: 8px;
  }
}

/* ===== Right Panel ===== */

.right-panel {
  overflow: hidden;
  min-width: 0;
  border-left: 1px solid hsl(var(--border));
  background-color: hsl(var(--background));
}

.right-panel[data-visible="false"] {
  visibility: hidden;
  pointer-events: none;
}

/* ===== Sidebar ===== */

.app-shell-sidebar {
  border-right: 1px solid hsl(var(--border));
  background-color: hsl(var(--sidebar) / 0.5);
}

/* ===== Theme Transition ===== */

/* Per D-09 and UI-SPEC: 300ms ease-out cross-fade on all color properties */
*,
*::before,
*::after {
  transition-property: background-color, color, border-color, box-shadow;
  transition-duration: 300ms;
  transition-timing-function: ease-out;
}

/* Instant transitions for elements that should NOT animate */
.app-shell,
.app-shell *[data-no-transition],
.no-theme-transition {
  transition-duration: 0ms !important;
}

/* Ensure grid changes still animate */
.app-shell {
  transition-property: grid-template-columns;
  transition-duration: 200ms;
  transition-timing-function: ease-out;
}

/* Prevent transition on initial page load (flash prevention) */
.preload * {
  transition: none !important;
}
```
