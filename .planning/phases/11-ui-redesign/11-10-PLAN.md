  <action>
Create web/app/sw.ts — the Serwist service worker entry file:

```typescript
import { defaultCache } from '@serwist/next/worker'
import type { PrecacheEntry, SerwistGlobalConfig } from 'serwist'
import { Serwist } from 'serwist'

declare global {
  interface WorkerGlobalScope extends SerwistGlobalConfig {
    __SW_MANIFEST: (PrecacheEntry | string)[] | undefined
  }
}

declare const self: ServiceWorkerGlobalScope & WorkerGlobalScope

const serwist = new Serwist({
  precacheEntries: self.__SW_MANIFEST,
  skipWaiting: true,
  clientsClaim: true,
  navigationPreload: false,
  runtimeCaching: defaultCache,
})

serwist.addEventListeners()
```

Also create a new service worker registration in web/app/layout.tsx (the root layout). Add this inside the `<body>` but before the provider tree — actually, it needs to be inside a 'use client' component. Since layout.tsx is a Server Component, use a separate client component for SW registration:

Create web/components/layout/ServiceWorkerRegister.tsx:

```typescript
'use client'

import { useEffect } from 'react'

export function ServiceWorkerRegister() {
  useEffect(() => {
    if ('serviceWorker' in navigator && process.env.NODE_ENV === 'production') {
      navigator.serviceWorker
        .register('/sw.js')
        .then((reg) => console.log('[SW] Registered:', reg.scope))
        .catch((err) => console.error('[SW] Registration failed:', err))
    }
  }, [])

  return null
}
```

Then in web/app/layout.tsx, add the import and render the component inside the body (inside ThemeProvider, after ThemeColorProvider closes):

```typescript
// Add this import
import { ServiceWorkerRegister } from '@/components/layout/ServiceWorkerRegister'

// Inside the provider tree body, after </ThemeColorProvider> and before </body>:
  {children}
</ThemeColorProvider>
<ServiceWorkerRegister />
```

Alternatively, since the plan explicitly says SW registration goes in the root layout, add it directly. But the layout IS a Server Component, so the SW registration MUST go through a client component wrapper.

The key: After the SW registration file is created, update Plan 04's already-created layout.tsx (or modify it now) to include <ServiceWorkerRegister />.
